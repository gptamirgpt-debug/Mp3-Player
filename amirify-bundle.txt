package com.amir.mp3player

import android.content.Context
import org.json.JSONObject

object Store {

    private fun prefs(c: Context) = c.getSharedPreferences("amirify", Context.MODE_PRIVATE)

    fun favorites(c: Context): Set<Long> {
        val raw: Set<String> = prefs(c).getStringSet("fav", emptySet()) ?: emptySet()
        return raw.mapNotNull { it.toLongOrNull() }.toSet()
    }

    fun toggleFavorite(c: Context, id: Long): Boolean {
        val current: Set<String> = prefs(c).getStringSet("fav", emptySet()) ?: emptySet()
        val set = HashSet<String>(current)
        val key = id.toString()
        val nowFav = if (set.contains(key)) {
            set.remove(key)
            false
        } else {
            set.add(key)
            true
        }
        prefs(c).edit().putStringSet("fav", set).apply()
        return nowFav
    }

    fun recordPlay(c: Context, id: Long) {
        val p = prefs(c)
        val counts = JSONObject(p.getString("counts", "{}") ?: "{}")
        val last = JSONObject(p.getString("last", "{}") ?: "{}")
        val k = id.toString()
        counts.put(k, counts.optInt(k, 0) + 1)
        last.put(k, System.currentTimeMillis())
        p.edit()
            .putString("counts", counts.toString())
            .putString("last", last.toString())
            .apply()
    }

    fun playCounts(c: Context): Map<Long, Int> {

package com.amir.mp3player

import android.content.Context
import android.media.MediaCodec
import android.media.MediaExtractor
import android.media.MediaFormat
import android.net.Uri
import java.io.File
import java.nio.ByteOrder

object WaveformExtractor {

    private fun file(ctx: Context, id: Long) = File(ctx.filesDir, "wave_" + id + ".bin")

    fun cached(ctx: Context, id: Long): FloatArray? {
        val f = file(ctx, id)
        if (!f.exists()) return null
        return try {
            val b = f.readBytes()
            if (b.isEmpty()) null else FloatArray(b.size) { (b[it].toInt() and 0xFF) / 255f }
        } catch (e: Exception) {
            null
        }
    }

    fun save(ctx: Context, id: Long, peaks: FloatArray) {
        try {
            file(ctx, id).writeBytes(
                ByteArray(peaks.size) { (peaks[it].coerceIn(0f, 1f) * 255f).toInt().toByte() }
            )
        } catch (e: Exception) {
        }
    }

    // A nice-looking placeholder wave shown until the real one is ready.
    fun pseudo(seed: Long, n: Int): FloatArray {
        val r = java.util.Random(seed)
        var prev = 0.5f
        return FloatArray(n) {
            prev = prev * 0.55f + (0.25f + r.nextFloat() * 0.75f) * 0.45f
            prev.coerceIn(0.12f, 1f)
        }
    }

    // Decodes the song in the background and returns the loudness of each of the [bars] slices.
    fun extract(ctx: Context, uri: Uri, bars: Int, cancelled: () -> Boolean): FloatArray? {
        val extractor = MediaExtractor()
        var codec: MediaCodec? = null
        try {
            extractor.setDataSource(ctx, uri, null)
            var track = -1
            var found: MediaFormat? = null
            for (i in 0 until extractor.trackCount) {
                val f = extractor.getTrackFormat(i)
                val m = f.getString(MediaFormat.KEY_MIME) ?: ""
                if (m.startsWith("audio/")) {
                    track = i
                    found = f
                    break
                }
            }
            val format = found ?: return null
            if (track < 0) return null
            if (!format.containsKey(MediaFormat.KEY_DURATION)) return null
            val durationUs = format.getLong(MediaFormat.KEY_DURATION)
            if (durationUs <= 0L) return null
            val mime = format.getString(MediaFormat.KEY_MIME) ?: return null
            extractor.selectTrack(track)

            val dec = MediaCodec.createDecoderByType(mime)
            codec = dec
            dec.configure(format, null, null, 0)
            dec.start()

            val peaks = FloatArray(bars)
            val info = MediaCodec.BufferInfo()
            var inputDone = false
            var outputDone = false
            var idle = 0

            while (!outputDone) {
                if (cancelled()) return null

                if (!inputDone) {
                    val inIdx = dec.dequeueInputBuffer(5000)
                    if (inIdx >= 0) {
                        val buf = dec.getInputBuffer(inIdx)
                        val size = if (buf != null) extractor.readSampleData(buf, 0) else -1
                        if (size < 0) {
                            dec.queueInputBuffer(inIdx, 0, 0, 0L, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                            inputDone = true
                        } else {
                            dec.queueInputBuffer(inIdx, 0, size, extractor.sampleTime, 0)
                            extractor.advance()
                        }
                    }
                }

                val outIdx = dec.dequeueOutputBuffer(info, 5000)
                if (outIdx >= 0) {
                    idle = 0
                    if (info.size > 0) {
                        val out = dec.getOutputBuffer(outIdx)
                        if (out != null) {
                            out.position(info.offset)
                            out.limit(info.offset + info.size)
                            val sb = out.order(ByteOrder.LITTLE_ENDIAN).asShortBuffer()
                            val total = sb.remaining()
                            var peak = 0
                            var k = 0
                            while (k < total) {
                                val v = Math.abs(sb.get(k).toInt())
                                if (v > peak) peak = v
                                k += 4
                            }
                            val bar = ((info.presentationTimeUs * bars) / durationUs)
                                .toInt().coerceIn(0, bars - 1)
                            val p = peak / 32768f
                            if (p > peaks[bar]) peaks[bar] = p
                        }
                    }
                    dec.releaseOutputBuffer(outIdx, false)
                    if ((info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) outputDone = true
                } else {
                    idle++
                    if (idle > 2000) return null
                }
            }

            var max = 0f
            for (p in peaks) if (p > max) max = p
            if (max <= 0f) return null
            for (i in peaks.indices) {
                peaks[i] = Math.sqrt((peaks[i] / max).toDouble()).toFloat().coerceIn(0.08f, 1f)
            }
            return peaks
        } catch (e: Exception) {
            return null
        } finally {
            try {
                codec?.stop()
            } catch (e: Exception) {
            }
            try {
                codec?.release()
            } catch (e: Exception) {
            }
            try {
                extractor.release()
            } catch (e: Exception) {
            }
        }
    }
}

package com.amir.mp3player

import android.content.Context
import android.graphics.Canvas
import android.graphics.Color
import android.graphics.Paint
import android.util.AttributeSet
import android.view.MotionEvent
import android.view.View

class WaveformView @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
    defStyleAttr: Int = 0
) : View(context, attrs, defStyleAttr) {

    var peaks: FloatArray = FloatArray(0)
        set(value) {
            field = value
            invalidate()
        }

    var progress: Float = 0f
        set(value) {
            field = value.coerceIn(0f, 1f)
            invalidate()
        }

    var onSeekStart: (() -> Unit)? = null
    var onSeekChange: ((Float) -> Unit)? = null
    var onSeekEnd: ((Float) -> Unit)? = null

    private val playedPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = Color.parseColor("#B388FF")
        strokeCap = Paint.Cap.ROUND
    }
    private val restPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = Color.parseColor("#4DFFFFFF")
        strokeCap = Paint.Cap.ROUND
    }
    private var dragging = false

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)
        val n = if (peaks.isEmpty()) 72 else peaks.size
        val w = width.toFloat()
        val h = height.toFloat()
        val slot = w / n
        val stroke = slot * 0.56f
        playedPaint.strokeWidth = stroke
        restPaint.strokeWidth = stroke
        val minHalf = stroke / 2f
        for (i in 0 until n) {
            val p = if (peaks.isEmpty()) 0.12f else peaks[i]
            val half = maxOf(minHalf, p * h * 0.5f * 0.92f)
            val x = slot * (i + 0.5f)
            val paint = if ((i + 0.5f) / n <= progress) playedPaint else restPaint
            canvas.drawLine(x, h / 2f - half, x, h / 2f + half, paint)
        }
    }

    override fun onTouchEvent(e: MotionEvent): Boolean {
        if (width == 0) return true
        val f = (e.x / width).coerceIn(0f, 1f)
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> {
                parent?.requestDisallowInterceptTouchEvent(true)
                dragging = true
                onSeekStart?.invoke()
                progress = f
                onSeekChange?.invoke(f)
            }
            MotionEvent.ACTION_MOVE -> {
                if (dragging) {
                    progress = f
                    onSeekChange?.invoke(f)
                }
            }
            MotionEvent.ACTION_UP -> {
                if (dragging) {
                    dragging = false
                    progress = f
                    onSeekEnd?.invoke(f)
                    performClick()
                }
            }
            MotionEvent.ACTION_CANCEL -> {
                if (dragging) {
                    dragging = false
                    onSeekEnd?.invoke(progress)
                }
            }
        }
        return true
    }

    override fun performClick(): Boolean {
        return super.performClick()
    }
}

    override fun onCreate() {
        super.onCreate()

            .setAudioAttributes(
                AudioAttributes.Builder()

                val id = mediaItem?.mediaId?.toLongOrNull() ?: return
                Store.recordPlay(applicationContext, id)

        })

    }

    override fun onGetSession(controllerInfo: MediaSession.ControllerInfo): MediaSession? = mediaSession

        val player = mediaSession?.player

            stopSelf()
        }
    }

        mediaSession?.run {
            player.release()
            release()
            mediaSession = null
        }
        super.onDestroy()
    }

        updateNowPlaying()
        updatePlayButton()
        updateModeButtons()
        updateProgress()
    }

    private fun updateNowPlaying() {
        val c = controller ?: return
        val item = c.currentMediaItem
        val id = item?.mediaId?.toLongOrNull() ?: -1L
        val song = songs.firstOrNull { it.id == id }

        binding.tvTitle.text = song?.title ?: "چیزی در حال پخش نیست"
        binding.tvArtist.text = song?.artist ?: "یه آهنگ انتخاب کن"
        adapter.setCurrent(id)

        if (song == null) {
            currentSongId = -1L
            binding.playerArt.setImageDrawable(null)
            binding.wave.peaks = FloatArray(0)
        } else if (song.id != currentSongId) {
            currentSongId = song.id
            ArtLoader.load(this, song, binding.playerArt, 160)
            loadWave(song)
        }
        updateHeart()
    }

    private fun loadWave(song: Song) {
        waveFor = song.id
        val cached = WaveformExtractor.cached(this, song.id)
        if (cached != null) {
            binding.wave.peaks = cached
            return
        }
        binding.wave.peaks = WaveformExtractor.pseudo(song.id, 72)
        val appCtx = applicationContext
        waveExecutor.execute {
            val real = WaveformExtractor.extract(appCtx, song.uri, 72) { waveFor != song.id }
            if (real != null) {
                WaveformExtractor.save(appCtx, song.id, real)
                runOnUiThread {
                    if (waveFor == song.id) binding.wave.peaks = real
                }
            }
        }
    }

    private fun updateHeart() {
        val fav = currentSongId >= 0 && favs.contains(currentSongId)
        binding.btnFav.setImageResource(if (fav) R.drawable.ic_heart else R.drawable.ic_heart_outline)
        binding.btnFav.imageTintList = ColorStateList.valueOf(
            ContextCompat.getColor(this, if (fav) R.color.pink else R.color.text_secondary)
        )
    }

    private fun updatePlayButton() {
        val c = controller ?: return
        binding.btnPlay.setImageResource(if (c.isPlaying) R.drawable.ic_pause else R.drawable.ic_play)
    }

    private fun updateModeButtons() {
        val c = controller ?: return
        val accent = ContextCompat.getColor(this, R.color.accent)
        val dim = ContextCompat.getColor(this, R.color.text_secondary)
        binding.btnShuffle.imageTintList = ColorStateList.valueOf(if (c.shuffleModeEnabled) accent else dim)
        when (c.repeatMode) {
            Player.REPEAT_MODE_ONE -> {
                binding.btnRepeat.setImageResource(R.drawable.ic_repeat_one)
                binding.btnRepeat.imageTintList = ColorStateList.valueOf(accent)
            }
            Player.REPEAT_MODE_ALL -> {
                binding.btnRepeat.setImageResource(R.drawable.ic_repeat)
                binding.btnRepeat.imageTintList = ColorStateList.valueOf(accent)
            }
            else -> {
                binding.btnRepeat.setImageResource(R.drawable.ic_repeat)
                binding.btnRepeat.imageTintList = ColorStateList.valueOf(dim)
            }
        }
    }

    private fun updateProgress() {
        val c = controller ?: return
        val dur = if (c.duration > 0) c.duration else 0L
        val pos = c.currentPosition.coerceAtLeast(0L)
        if (!userSeeking) {
            binding.wave.progress = if (dur > 0) pos.toFloat() / dur.toFloat() else 0f
            binding.tvElapsed.text = formatTime(pos)
        }
        binding.tvTotal.text = if (showRemaining) {
            "-" + formatTime((dur - pos).coerceAtLeast(0L))
        } else {
            formatTime(dur)
        }
    }

    private fun formatTime(ms: Long): String {
        val total = ms / 1000
        return String.format(Locale.US, "%d:%02d", total / 60, total % 60)
    }
}
